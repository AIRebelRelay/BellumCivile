    public sealed class TextObject {
        public string Value;public bool ThrowOnCopy,ThrowOnRender;public int RenderCalls;
        public Dictionary<string,string> Variables=new Dictionary<string,string>();
        public TextObject(string value){Value=value;}
        public TextObject CopyTextObject(){if(ThrowOnCopy)throw new InvalidOperationException("fixture copy failed");return new TextObject(Value){ThrowOnRender=ThrowOnRender,Variables=new Dictionary<string,string>(Variables)};}
        public override string ToString(){RenderCalls++;if(ThrowOnRender||Value=="THROW")throw new InvalidOperationException("fixture render failed");string s=Value;if(s=="{=town}Fallback")return "Rovalt";foreach(var p in Variables)s=s.Replace("{"+p.Key+"}",p.Value);return s;}
    }
}
