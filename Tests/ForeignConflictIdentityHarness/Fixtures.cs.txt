public static class IdentityReaderDigest{
 public static string Dump(object value){
  if(value==null)return "null";Type t=value.GetType();if(t.IsPrimitive||t.IsEnum||t==typeof(string))return t.FullName+":"+Convert.ToString(value,System.Globalization.CultureInfo.InvariantCulture);
  var list=value as IList;if(list!=null)return "["+string.Join(";",list.Cast<object>().Select(Dump))+"]";
  var map=value as IDictionary;if(map!=null){var rows=new List<string>();foreach(DictionaryEntry p in map)rows.Add(Dump(p.Key)+"="+Dump(p.Value));return "{"+string.Join(";",rows.OrderBy(x=>x))+"}";}
  return string.Join(";",t.GetFields(BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic).Where(f=>f.GetCustomAttribute<TaleWorlds.SaveSystem.SaveableFieldAttribute>()!=null).OrderBy(f=>f.Name).Select(f=>f.Name+"="+Dump(f.GetValue(value))));
 }
}
