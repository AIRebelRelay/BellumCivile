namespace TaleWorlds.CampaignSystem {
    public sealed class Campaign {
        public static Campaign Current = new Campaign();
        public BellumCivile.Behaviors.FeudalTitleBehavior Titles;
        public BellumCivile.Behaviors.BellumIntegrationBehavior Integration = new BellumCivile.Behaviors.BellumIntegrationBehavior();
        public T GetCampaignBehavior<T>() where T : class { return Titles as T; }
    }
    public sealed class CampaignTime { public static CampaignTime Now = new CampaignTime(); public double ToDays = 100; }
    public sealed class Hero { public string StringId; }
    public sealed class Clan { public static List<Clan> All = new List<Clan>(); public string StringId; public bool IsEliminated; }
}

    public sealed partial class BellumIntegrationBehavior {
        internal static BellumIntegrationBehavior Current { get { return Campaign.Current == null ? null : Campaign.Current.Integration; } }

        private HashSet<string> _illegitimateIndex = new HashSet<string>(StringComparer.Ordinal);
        public void Bar(string id) { _illegitimateIndex.Add(id); }
    }
    public partial class FeudalTitleBehavior {
        private List<BellumCivile.FeudalClaimRecord> _claims = new List<BellumCivile.FeudalClaimRecord>();
